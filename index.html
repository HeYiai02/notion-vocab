async function handleAnswer(isRemembered) { const word = dueList[currentIndex]; // 锁定按钮，防止重复点击 document.getElementById('btn-forgot').disabled = true; document.getElementById('btn-remember').disabled = true; if(isRemembered) document.getElementById('btn-remember').innerText = "保存中..."; else document.getElementById('btn-forgot').innerText = "保存中..."; let newLevel; if (isRemembered) { // 记住了：等级提升 newLevel = Math.min(word.level + 1, INTERVALS.length - 1); } else { // 忘记了：等级打回原形 newLevel = 0; // 【核心修复】：将这个忘记的单词克隆一份，把新等级设为 0，然后塞回队列末尾！ dueList.push({ ...word, level: 0 }); } const daysToAdd = INTERVALS[newLevel]; const nextDate = new Date(Date.now() + daysToAdd * 24 * 60 * 60 * 1000); const nextReviewDate = nextDate.toISOString().split('T')[0]; // 提交云端更新 Notion await fetch('/.netlify/functions/update_word', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ pageId: word.pageId, newLevel: newLevel, nextReviewDate: nextReviewDate }) }); // 进入下一张卡片 currentIndex++; showCard(); }
